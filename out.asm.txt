section .data
fmt_int: db "%ld", 10, 0
fmt_read: db "%ld", 0

section .bss
mem: resq 6

section .text
global main
extern printf, scanf, puts

main:
    push rbp
    mov rbp, rsp
    mov rax, 3
    mov [rel mem + 8], rax
    mov rax, 6
    mov [rel mem + 16], rax
    mov rax, [rel mem + 16]
    mov [rel mem + 24], rax
    mov rsi, [rel mem + 24]
    lea rdi, [rel fmt_int]
    xor rax, rax
    call printf
    lea rdi, [rel fmt_read]
    lea rsi, [rel mem + 8]
    xor rax, rax
    call scanf
    mov rax, [rel mem + 8]
    mov rbx, [rel mem + 16]
    add rax, rbx
    mov [rel mem + 40], rax
    mov rsi, [rel mem + 40]
    lea rdi, [rel fmt_int]
    xor rax, rax
    call printf
    mov rsp, rbp
    pop rbp
    ret
